# -----------------------------------------------------
# EC2 インスタンスに boto3 をインストールする
# -----------------------------------------------------

# ec2-user に切り替える
sudo su ec2-user

# ホームディレクトリに移動する
cd ~

# Python 3.14 をインストールする
sudo dnf install python3.14 -y

# pip を利用できるようにする
python3.14 -m ensurepip --upgrade

# boto3 をインストールする
python3.14 -m pip install boto3

# -----------------------------------------------------
# EC2 インスタンス名から対象のインスタンス ID を取得する
# -----------------------------------------------------

# Python ファイルを作成する
vim create_image.py

# Python ファイルを実行する
python3.14 create_image.py

# -----------------------------------------------------
# GitHub のサンプルコードをそのまま使う場合
# -----------------------------------------------------

# ホームディレクトリに移動する
cd ~

# Git をインストールする
sudo dnf install -y git

# GitHub リポジトリを取得する
git clone https://github.com/ketancho/aws-hands-on-book-resources.git

# Day15 のサンプルコードをコピーする
cp aws-hands-on-book-resources/assets/Day15/create_image-v1.py create_image.py

# Python ファイルを実行する
python3.14 create_image.py
