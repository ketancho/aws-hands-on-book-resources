# -----------------------------------------------------
# EC2インスタンスにboto3をインストールする
# -----------------------------------------------------

# ec2-userに切り替える
sudo su ec2-user

# ホームディレクトリに移動する
cd ~

# Python 3.14をインストールする
sudo dnf install python3.14 -y

# pipを利用できるようにする
python3.14 -m ensurepip --upgrade

# boto3をインストールする
python3.14 -m pip install boto3

# -----------------------------------------------------
# EC2インスタンス名から対象のインスタンスIDを取得する
# -----------------------------------------------------

# Pythonファイルを作成する
vim create_image.py

# Pythonファイルを実行する
python3.14 create_image.py

# -----------------------------------------------------
# GitHubのサンプルコードをそのまま使う場合
# -----------------------------------------------------

# ホームディレクトリに移動する
cd ~

# Gitをインストールする
sudo dnf install -y git

# GitHubリポジトリを取得する
git clone https://github.com/ketancho/aws-hands-on-book-resources.git

# Day15のサンプルコードをコピーする
cp aws-hands-on-book-resources/assets/Day15/create_image-v1.py create_image.py

# Pythonファイルを実行する
python3.14 create_image.py
