# MySQL Community Repositoryを追加する
sudo dnf -y install https://dev.mysql.com/get/mysql84-community-release-el9-1.noarch.rpm

# MySQLクライアントとPHPドライバをインストールする
sudo dnf -y install mysql php-mysqlnd

# RDSインスタンスに接続する
mysql -h <RDSエンドポイント> -u admin -p

# データベースを作成する

# データベースを選択する
USE aws_learning_journal;

# journalsテーブルを作成する
CREATE TABLE journals (
  id INT NOT NULL PRIMARY KEY,
  title VARCHAR(100),
  learning VARCHAR(1000),
  image VARCHAR(1000)
);
