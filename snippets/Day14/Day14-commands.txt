# =====================================================
# Day14: AWS CLI
# =====================================================

# -----------------------------------------------------
# バッチインスタンスの現状を確認する
# -----------------------------------------------------

# ec2-user に切り替える
sudo su ec2-user

# ホームディレクトリに移動する
cd ~

# AWS CLI のバージョンを確認する
aws --version

# EC2 インスタンス一覧を取得する
aws ec2 describe-instances

# -----------------------------------------------------
# query / filters / output オプションを利用する
# -----------------------------------------------------

# EC2 インスタンス名・インスタンス ID・ステータスを表示する
aws ec2 describe-instances \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]'

# Name タグで EC2 インスタンスを絞り込む
aws ec2 describe-instances \
--filters "Name=tag:Name,Values=aws-hands-on-book-web-1a" \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]'

# 表形式で表示する
aws ec2 describe-instances \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]' \
--output table

# -----------------------------------------------------
# AWS CLI を使って web-1a インスタンスを停止する
# -----------------------------------------------------

# web-1a インスタンスを停止する
aws ec2 stop-instances --instance-ids <aws-hands-on-book-web-1a のインスタンス ID を入力>

# -----------------------------------------------------
# AWS CLI を使って web-1a インスタンスを再開する
# -----------------------------------------------------

# web-1a インスタンスを開始する
aws ec2 start-instances --instance-ids <aws-hands-on-book-web-1a のインスタンス ID を入力>
