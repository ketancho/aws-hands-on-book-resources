# =====================================================
# Day14: AWS CLI
# =====================================================

# -----------------------------------------------------
# バッチインスタンスの現状を確認する
# -----------------------------------------------------

# ec2-userに切り替える
sudo su ec2-user

# ホームディレクトリに移動する
cd ~

# AWS CLIのバージョンを確認する
aws --version

# EC2インスタンス一覧を取得する
aws ec2 describe-instances

# -----------------------------------------------------
# query/filters/outputオプションを利用する
# -----------------------------------------------------

# EC2インスタンス名・インスタンスID・ステータスを表示する
aws ec2 describe-instances \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]'

# NameタグでEC2インスタンスを絞り込む
aws ec2 describe-instances \
--filters "Name=tag:Name,Values=aws-hands-on-book-web-1a" \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]'

# 表形式で表示する
aws ec2 describe-instances \
--query 'Reservations[*].Instances[*].[Tags[?Key==`Name`].Value|[0],InstanceId,State.Name]' \
--output table

# -----------------------------------------------------
# AWS CLIを使ってweb-1aインスタンスを停止する
# -----------------------------------------------------

# web-1aインスタンスを停止する
aws ec2 stop-instances --instance-ids <aws-hands-on-book-web-1aのインスタンスIDを入力>

# -----------------------------------------------------
# AWS CLIを使ってweb-1aインスタンスを再開する
# -----------------------------------------------------

# web-1aインスタンスを開始する
aws ec2 start-instances --instance-ids <aws-hands-on-book-web-1aのインスタンスIDを入力>
