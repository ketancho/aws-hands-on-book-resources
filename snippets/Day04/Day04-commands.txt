# =====================================================
# Day4: Amazon EC2
# =====================================================

# -----------------------------------------------------
# EC2インスタンスにSSH接続する
# -----------------------------------------------------

# .pemキーの権限を変更する
chmod 400 aws-hands-on-book.pem

# EC2インスタンスにSSH接続する
ssh -i aws-hands-on-book.pem ec2-user@<EC2インスタンスのパブリックIPv4アドレス>

# CloudShellのIPアドレスを確認する
curl http://checkip.amazonaws.com/

# -----------------------------------------------------
# EC2インスタンスをWebサーバーにする
# -----------------------------------------------------

# ソフトウェアを最新の状態に更新する
sudo dnf update -y

# Apache HTTP Server（httpd）をインストールする
sudo dnf install -y httpd

# httpdを起動する
sudo systemctl start httpd

# EC2インスタンス起動時に自動でhttpdを起動する
sudo systemctl enable httpd

# -----------------------------------------------------
# アプリケーションをデプロイする
# -----------------------------------------------------

# PHPをインストールする
sudo dnf install -y php8.5

# httpd.confを修正する
sudo vim /etc/httpd/conf/httpd.conf

# 設定ファイルの文法を確認する
httpd -t

# httpdを再起動する
sudo systemctl restart httpd

# Gitをインストールする
sudo dnf install -y git

# GitHubリポジトリを取得する
git clone https://github.com/ketancho/aws-hands-on-book-resources.git

# Webアプリケーションを配置する
sudo cp -r aws-hands-on-book-resources/assets/Day04/* /var/www/html/
