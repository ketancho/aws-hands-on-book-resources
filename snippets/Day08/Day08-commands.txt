# -----------------------------------------------------
# 画像をCloudFront経由で取得するようデータベースをアップデートする
# -----------------------------------------------------

# RDSインスタンスに接続する
mysql -h <RDSエンドポイント> -u admin -p

# データベースを選択する
USE aws_learning_journal;

# image列をCloudFrontのURLに更新する
UPDATE journals
SET image = REPLACE(
    image,
    'https://<S3バケット名>.s3.ap-northeast-1.amazonaws.com',
    'https://<CloudFrontディストリビューションドメイン名>'
);
