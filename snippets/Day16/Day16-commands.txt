# =====================================================
# Day16: S3 Static Website Hosting
# =====================================================

# -----------------------------------------------------
# S3バケットにサンプルアプリケーションを配置する
# -----------------------------------------------------

# GitHubリポジトリを取得する
git clone https://github.com/ketancho/aws-hands-on-book-resources.git

# 作業ディレクトリを作成する
mkdir serverless-app-work

# サンプルアプリケーションをコピーする
cp -r aws-hands-on-book-resources/assets/Day16/s3/* serverless-app-work/

# S3バケットへ同期する
aws s3 sync serverless-app-work s3://<本章Day16で作成したS3バケット名>

# -----------------------------------------------------
# アプリケーションを修正し、S3バケットに配置する
# -----------------------------------------------------

# index.htmlを編集する
vim serverless-app-work/index.html

# S3バケットへ同期する
aws s3 sync serverless-app-work s3://<本章Day16で作成したS3バケット名>
