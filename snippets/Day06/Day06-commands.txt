# -----------------------------------------------------
# AMIから2台目のEC2インスタンスを作成する
# -----------------------------------------------------

# EC2インスタンスweb-1aにSSH接続する
ssh -i aws-hands-on-book.pem ec2-user@<web-1aインスタンスのパブリックIPv4アドレス>

# index.phpを修正する
sudo vim /var/www/html/index.php

# CloudShellに戻る
exit

# EC2インスタンスweb-1cにSSH接続する
ssh -i aws-hands-on-book.pem ec2-user@<web-1cインスタンスのパブリックIPv4アドレス>

# index.phpを修正する
sudo vim /var/www/html/index.php

# CloudShellに戻る
exit

# -----------------------------------------------------
# 片方のEC2インスタンスに障害を発生させたときの挙動を確認する
# -----------------------------------------------------

# web-1cのhttpdを停止する
sudo systemctl stop httpd

# web-1cのhttpdを起動する
sudo systemctl start httpd
